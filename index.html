# H 代币经济模型计算器 - 严格基于你的数据 # Python 3 运行（无需安装包） lp_h = 17351598.0 # LP 中的 H lp_u = 110000.0 # LP 中的 USDT community_h = 682648402.0 staking_pool = 300000000.0 # 奖励池 treasury_h = 0.0 total_supply = 1000000000.0 k = lp_h * lp_u # 恒积 def get_daily_rate(days): if 1 <= days <= 7: return 0.005 elif 8 <= days <= 16: return 0.006 elif 17 <= days <= 30: return 0.007 elif 31 <= days <= 60: return 0.008 else: return 0.01 def buy_and_stake(usdt_in, days): global lp_h, lp_u, staking_pool, treasury_h, k if usdt_in <= 0 or days < 1: print("输入无效！") return # 1. Uniswap 买入（恒积公式，无滑点税） new_u = lp_u + usdt_in new_h = k / new_u h_out = lp_h - new_h # 2. 5% 交易税（H 进入国库） h_received = h_out * 0.95 tax_h = h_out * 0.05 treasury_h += tax_h # 更新 LP lp_h = new_h lp_u = new_u k = lp_h * lp_u current_price = lp_u / lp_h # 3. 质押 & 计算收益 stake_amount = h_received rate = get_daily_rate(days) total_reward = stake_amount * rate * days # 4. 扣奖励池 if staking_pool >= total_reward: staking_pool -= total_reward sustainable = "✅ 可持续" else: sustainable = "❌ 不可持续（奖励池不足！建议社区国库补充）" print("\n=== 模拟结果 ===") print(f"买入 {usdt_in} USDT → 获得 H: {h_received:,.2f}（税后）") print(f"新价格: {current_price:.6f} U/H") print(f"质押量: {stake_amount:,.2f} H | 锁定期: {days} 天 | 日收益: {rate*100:.1f}%") print(f"本次总奖励: {total_reward:,.2f} H（从质押池扣除）") print(f"剩余奖励池: {staking_pool:,.2f} H") print(f"国库累积: {treasury_h:,.2f} H") print(f"可持续性: {sustainable}") print("当前总量:") print(f" LP: {lp_h:,.0f} H + {lp_u:,.0f} USDT") print(f" 社区持币: {community_h:,.0f} H") print(f" 质押池: {staking_pool:,.0f} H") print(f" 总供应: {total_supply:,.0f} H") def show_current(): price = lp_u / lp_h print("\n=== 当前状态 ===") print(f"LP: {lp_h:,.0f} H + {lp_u:,.0f} USDT | 价格: {price:.6f} U/H") print(f"社区持币: {community_h:,.0f} H") print(f"质押池(奖励): {staking_pool:,.0f} H (剩余 {staking_pool/total_supply*100:.1f}%)") print(f"国库: {treasury_h:,.0f} H") print(f"可持续性: {'✅ 可持续' if staking_pool >= 0 else '❌ 警告'}") # 交互菜单 print("H 代币经济模型计算器启动！（基于你的全部数据）") while True: print("\n菜单：") print("1. 显示当前总量 & 可持续性") print("2. 模拟买入USDT → 质押H（输入金额+天数）") print("3. 退出") choice = input("输入选项 (1/2/3): ") if choice == "1": show_current() elif choice == "2": try: usdt = float(input("输入买入 USDT 金额: ")) days = int(input("输入锁定期天数: ")) buy_and_stake(usdt, days) except: print("输入错误，请重试！") elif choice == "3": print("计算器退出。数据可持续性取决于实际质押量/买入量，建议实时监控奖励池！") break else: print("无效选项！")
